#ifndef __STUDENT_H_
#define __STUDENT_H_

#include "segmented_vector.h"

/* =====================================================================
   Segmented Vector - student.h

   [ โจทย์ ]
   CP::SegmentedVector คือ vector ที่เก็บข้อมูลแบบ "แบ่งเป็น block"
   แทนที่จะใช้อาเรย์ชั้นเดียว (T*) เหมือน CP::vector เดิม เราเปลี่ยน
   mData เป็นอาเรย์สองมิติ (T**) โดย
     - mData[b]      คือ block ที่ b (เป็นอาเรย์ขนาด blockSize ช่อง)
     - blockSize     คือ ขนาดของแต่ละ block (มิติที่ 2)
     - mCap          คือ จำนวน block ที่จองไว้ (มิติที่ 1) *ไม่ใช่* จำนวนช่องทั้งหมด
     - mSize         คือ จำนวนข้อมูลจริงที่เก็บอยู่ (นับรวมทุก block)

   ข้อมูลถูกเก็บแบบ "เรียงต่อกันไม่มีช่องว่าง" (contiguous ในเชิงตรรกะ)
   นั่นคือ ข้อมูลลำดับที่ index จะอยู่ที่ block ที่ index / blockSize
   และอยู่ในช่องที่ index % blockSize ของ block นั้น
   ทำให้การเข้าถึงด้วย index ยังคงมองเหมือน vector ปกติทุกประการ

   งานที่ต้องเขียนในไฟล์นี้มี 4 ฟังก์ชัน คือ destructor, expand,
   insert และ operator[]

   [ แนวคิดหลักและเหตุผลที่ถูกต้อง ]
   1) operator[] : ใช้สูตรแปลง index เดียว (1 มิติ) ให้เป็นพิกัด 2 มิติ
      ด้วยการหาร/มอดุโล ตามที่อธิบายข้างบน คิดเป็น O(1)

   2) expand : จุดเด่นของ segmented vector คือ "ไม่ต้องย้ายข้อมูลจริง"
      เวลาขยาย เพราะข้อมูลถูกเก็บกระจายอยู่ในแต่ละ block อยู่แล้ว
      เราจึงจองเฉพาะอาเรย์ของ "พอยน์เตอร์" (มิติที่ 1) ขึ้นใหม่
      แล้วคัดลอกเฉพาะค่าพอยน์เตอร์ของ block เดิมมาใส่ (block เดิม
      ยังคงอยู่ที่ตำแหน่งเดิมในหน่วยความจำ ไม่ถูกลบและไม่ถูกคัดลอกข้อมูล)
      จากนั้นจึงจอง block ใหม่ให้ช่องที่เพิ่มขึ้นมา
      สุดท้าย delete [] เฉพาะอาเรย์พอยน์เตอร์ตัวเก่าเท่านั้น
      *ห้าม* delete block เดิม เพราะยังถูกใช้งานต่อผ่านพอยน์เตอร์ที่คัดลอกไว้
      (ถ้าเผลอลบจะกลายเป็น dangling pointer ทันที)

   3) insert : ต้องคำนวณก่อนว่าถ้าเพิ่มข้อมูลอีก 1 ตัว (รวมเป็น mSize+1 ตัว)
      จะต้องใช้กี่ block คำตอบคือ ceil((mSize+1) / blockSize) ซึ่งเขียน
      ด้วยเลขจำนวนเต็มได้เป็น (mSize + 1 + blockSize - 1) / blockSize
      แล้วส่งค่านี้ให้ ensureCapacity (ที่มีให้แล้วใน segmented_vector.h)
      เป็นผู้ตัดสินใจว่าจะ expand หรือไม่ และจะขยายเป็นเท่าไร
      *สำคัญ* : ต้องส่งเป็น "จำนวน block" ไม่ใช่ "จำนวนช่องข้อมูล"
      เพราะนิยามของ mCap ในคลาสนี้คือจำนวน block
      หลังจากมีที่พอแล้ว จึงเลื่อนข้อมูลตั้งแต่ตำแหน่งท้ายสุดถอยหลัง
      ลงมาจนถึงตำแหน่ง index ไปทางขวา 1 ช่อง แล้วค่อยวางข้อมูลใหม่ลงไป
      การเลื่อนต้องไล่จากหลังมาหน้าเสมอ ไม่งั้นข้อมูลจะถูกเขียนทับหายไป
      ตรงนี้เราเรียกใช้ operator[] ได้เลย ทำให้ไม่ต้องกังวลเรื่องข้ามรอย
      ต่อระหว่าง block (operator[] จัดการให้แล้ว) คิดเป็น O(mSize)

   4) destructor : ต้องคืนหน่วยความจำให้ครบทั้ง 2 ระดับ คือ ลบทุก block
      (มิติที่ 2) ก่อน แล้วจึงลบอาเรย์ของพอยน์เตอร์ (มิติที่ 1)
      ถ้าลบอาเรย์ชั้นนอกก่อน จะเข้าถึงพอยน์เตอร์ของ block ไม่ได้อีกเลย
      หมายเหตุ : ทุก block ตั้งแต่ 0 ถึง mCap-1 ถูกจองไว้แล้วเสมอ
      (constructor จอง block 0 ให้ และ expand จองส่วนที่เพิ่มมาให้ครบ)
      จึงวนลบได้ตรง ๆ ทั้ง mCap ช่อง
   ===================================================================== */

template <typename T>
CP::SegmentedVector<T>::~SegmentedVector()
{
    // คืนหน่วยความจำของ block ย่อยทุกอันก่อน (มิติที่ 2)
    // ทุก block ในช่วง [0, mCap) ถูกจองไว้แล้วเสมอ จึงลบได้ทั้งหมด
    for (size_t i = 0; i < mCap; i++) {
        delete [] mData[i];
    }
    // แล้วจึงคืนอาเรย์ของพอยน์เตอร์ (มิติที่ 1) เป็นลำดับสุดท้าย
    delete [] mData;
}

template <typename T>
void CP::SegmentedVector<T>::expand(size_t capacity)
{
    // กันกรณีถูกเรียกด้วยค่าที่ไม่ได้ทำให้ใหญ่ขึ้นจริง จะได้ไม่ทำข้อมูลหาย
    if (capacity <= mCap) return;

    // จองอาเรย์ของพอยน์เตอร์ชุดใหม่ตามจำนวน block ที่ต้องการ
    T** newData = new T*[capacity]();

    // ย้าย "พอยน์เตอร์" ของ block เดิมมาใช้ต่อ ไม่ได้คัดลอกตัวข้อมูล
    // จึงเร็วกว่าการ expand ของ vector ปกติที่ต้องคัดลอกข้อมูลทุกตัว
    for (size_t i = 0; i < mCap; i++) {
        newData[i] = mData[i];
    }

    // block ส่วนที่เพิ่มขึ้นมาใหม่ ต้องจองพื้นที่ให้ครบทุกอัน
    // เพื่อให้ค่าคงที่ (invariant) ที่ว่า "ทุก block ใน [0, mCap) ถูกจองแล้ว"
    // ยังคงเป็นจริง ทั้ง insert และ destructor พึ่งพาเงื่อนไขนี้
    for (size_t i = mCap; i < capacity; i++) {
        newData[i] = new T[blockSize]();
    }

    // ลบเฉพาะอาเรย์พอยน์เตอร์ตัวเก่าเท่านั้น ห้ามลบตัว block เดิม
    // เพราะ block เดิมยังถูกอ้างถึงอยู่ใน newData
    delete [] mData;
    mData = newData;
    mCap = capacity;
}

template <typename T>
void CP::SegmentedVector<T>::insert(int index, const T& element)
{
    // จำนวน block ที่ต้องใช้เมื่อมีข้อมูล mSize+1 ตัว = ceil((mSize+1)/blockSize)
    // เขียนด้วยเลขจำนวนเต็มเป็น (mSize + 1 + blockSize - 1) / blockSize
    size_t neededBlock = (mSize + 1 + blockSize - 1) / blockSize;
    ensureCapacity(neededBlock);

    // เลื่อนข้อมูลจากท้ายสุดถอยกลับมาถึงตำแหน่ง index ไปทางขวา 1 ช่อง
    // ต้องไล่จากหลังมาหน้า มิฉะนั้นข้อมูลจะถูกเขียนทับกันเอง
    // operator[] จัดการเรื่องการข้าม block ให้อยู่แล้ว จึงเขียนเหมือน vector ปกติ
    for (int i = (int)mSize; i > index; i--) {
        (*this)[i] = (*this)[i - 1];
    }

    // วางข้อมูลใหม่ลงตำแหน่งที่ว่างไว้ แล้วค่อยเพิ่มจำนวนข้อมูล
    (*this)[index] = element;
    mSize++;
}

template <typename T>
T& CP::SegmentedVector<T>::operator[](int index) const
{
    // แปลง index แบบ 1 มิติ ให้เป็นพิกัด 2 มิติของ mData
    // หาร = อยู่ block ไหน , มอดุโล = อยู่ช่องที่เท่าไรใน block นั้น
    int blockIndex = index / (int)blockSize;
    int offsetInBlock = index % (int)blockSize;
    return mData[blockIndex][offsetInBlock];
}
